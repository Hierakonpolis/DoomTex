%% DoomTex :: engine/map.tex
%% Level geometry.  24x24 grid, one digit per cell:
%%   0 = empty floor      1 = brick wall     2 = tech panel
%%   3 = stone wall       4 = door           5 = exit switch
%% Cells are expanded into control sequences at load time so that the
%% raycaster's cell lookup is a hash probe, not a string walk.
\ExplSyntaxOn

\int_const:Nn \c_dtex_map_w_int {24}
\int_const:Nn \c_dtex_map_h_int {24}

\int_new:N \l_dtex_row_x_int

%% \dtexMapRow {y} {24 digits}
\cs_new_protected:Npn \dtexMapRow #1#2
  {
    \int_zero:N \l_dtex_row_x_int
    \tl_map_inline:nn {#2}
      {
        \cs_gset:cpn { dtex_cell_ \int_use:N \l_dtex_row_x_int _ \int_eval:n {#1} : } {##1}
        \int_incr:N \l_dtex_row_x_int
      }
  }

%% Expandable cell lookup.  Anything off-grid reads as solid brick so the
%% raycaster can never walk off the edge of the world.
\cs_new:Npn \dtexCell #1#2
  {
    \cs_if_exist_use:cF { dtex_cell_ \int_eval:n {#1} _ \int_eval:n {#2} : } {1}
  }

%% A cell is solid unless it is empty floor, or a door that stands open.
\cs_new:Npn \dtexSolid #1#2
  {
    \int_case:nnF { \dtexCell {#1} {#2} }
      { {0} {0} {4} { \dtexDoorOpen {#1} {#2} } } {1}
  }

%% Doors are keyed by cell.  \dtexDoorOpen expands to 0 when open (i.e. not
%% solid) and 1 when shut.  Door state is carried in state.tex.
\cs_new:Npn \dtexDoorOpen #1#2
  { \cs_if_exist_use:cF { dtex_door_ \int_eval:n {#1} _ \int_eval:n {#2} : } {1} }

\cs_new_protected:Npn \dtexSetDoor #1#2#3 % x, y, 0=open 1=shut
  { \cs_gset:cpn { dtex_door_ \int_eval:n {#1} _ \int_eval:n {#2} : } {#3} }
