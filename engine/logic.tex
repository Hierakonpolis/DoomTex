%% DoomTex :: engine/logic.tex
%% One game tick: read the player's intent, move them, run the actors,
%% resolve combat.  Everything here is integer fixed-point.
\ExplSyntaxOn

\int_new:N \l_dtex_dx_int    \int_new:N \l_dtex_dy_int
\int_new:N \l_dtex_try_int   \int_new:N \l_dtex_marg_int
\int_new:N \l_dtex_spd_int   \int_new:N \l_dtex_turn_int
\int_new:N \l_dtex_ex_int    \int_new:N \l_dtex_ey_int
\int_new:N \l_dtex_fwd_int   \int_new:N \l_dtex_lat_int
\int_new:N \l_dtex_ed_int    \int_new:N \l_dtex_best_int
\int_new:N \l_dtex_bestd_int \int_new:N \l_dtex_los_int
\int_new:N \l_dtex_lsteps_int
\int_new:N \l_dtex_lx_int    \int_new:N \l_dtex_ly_int
\int_new:N \l_dtex_dmg_int   \int_new:N \l_dtex_tmp_int

\int_const:Nn \c_dtex_walk_int   {170}   % world units per frame, walking
\int_const:Nn \c_dtex_run_int    {300}
\int_const:Nn \c_dtex_turn_int   {24}    % angle units per frame (~8.4 deg)
\int_const:Nn \c_dtex_turnrun_int{44}
\int_const:Nn \c_dtex_radius_int {230}   % player collision radius
\int_const:Nn \c_dtex_imp_radius_int {380}

%% ------------------------------------------------------------------
%% Line of sight: march from (#1,#2) to (#3,#4) in eighth-unit steps and
%% report 1 if nothing solid blocks the way.
%% ------------------------------------------------------------------
\cs_new_protected:Npn \dtexLOSSet #1#2#3#4#5 % result, x1,y1,x2,y2
  {
    \int_set:Nn \l_dtex_dx_int { (#4) - (#2) }
    \int_set:Nn \l_dtex_dy_int { (#5) - (#3) }
    \dtexSqrtSet \l_dtex_ed_int
      { ( \l_dtex_dx_int * \l_dtex_dx_int + \l_dtex_dy_int * \l_dtex_dy_int ) / 1024 }
    \int_set:Nn #1 {1}
    \int_compare:nNnT { \l_dtex_ed_int } > {128}
      {
        % NOTE: the step count must live in its own variable.  Callers pass
        % their result variable as #1, and if that were also the divisor
        % here, the first blocked step would zero it and the next iteration
        % would divide by zero.
        \int_set:Nn \l_dtex_lsteps_int { \dtexMin { \l_dtex_ed_int / 128 } {96} }
        \int_step_inline:nnn {1} { \l_dtex_lsteps_int - 1 }
          {
            \int_set:Nn \l_dtex_lx_int
              { (#2) + \l_dtex_dx_int * (##1) / \l_dtex_lsteps_int }
            \int_set:Nn \l_dtex_ly_int
              { (#3) + \l_dtex_dy_int * (##1) / \l_dtex_lsteps_int }
            \int_compare:nNnT
              { \dtexSolid { \dtexFloor{\l_dtex_lx_int} } { \dtexFloor{\l_dtex_ly_int} } } > {0}
              { \int_set:Nn #1 {0} }
          }
      }
  }

%% ------------------------------------------------------------------
%% Movement with wall sliding: the two axes are resolved independently, so
%% walking into a wall at an angle slides along it instead of stopping dead.
%% ------------------------------------------------------------------
\cs_new_protected:Npn \dtexTryMove #1#2 % dx, dy
  {
    \int_compare:nNnF {#1} = {0}
      {
        \int_set:Nn \l_dtex_try_int { \dtexV{px} + (#1) }
        \int_set:Nn \l_dtex_marg_int
          { \int_compare:nNnTF {#1} > {0} { \c_dtex_radius_int } { 0-\c_dtex_radius_int } }
        \int_compare:nNnF
          { \dtexSolid { \dtexFloor { \l_dtex_try_int + \l_dtex_marg_int } }
                       { \dtexFloor { \dtexV{py} } } } > {0}
          { \dtexS {px} { \l_dtex_try_int } }
      }
    \int_compare:nNnF {#2} = {0}
      {
        \int_set:Nn \l_dtex_try_int { \dtexV{py} + (#2) }
        \int_set:Nn \l_dtex_marg_int
          { \int_compare:nNnTF {#2} > {0} { \c_dtex_radius_int } { 0-\c_dtex_radius_int } }
        \int_compare:nNnF
          { \dtexSolid { \dtexFloor { \dtexV{px} } }
                       { \dtexFloor { \l_dtex_try_int + \l_dtex_marg_int } } } > {0}
          { \dtexS {py} { \l_dtex_try_int } }
      }
  }

\cs_new_protected:Npn \dtexPlayerMove
  {
    \int_set:Nn \l_dtex_spd_int
      { \int_compare:nNnTF { \dtexIn{run} } = {1} { \c_dtex_run_int } { \c_dtex_walk_int } }
    \int_set:Nn \l_dtex_turn_int
      { \int_compare:nNnTF { \dtexIn{run} } = {1}
          { \c_dtex_turnrun_int } { \c_dtex_turn_int } }

    \int_compare:nNnT { \dtexIn{left} }  = {1}
      { \dtexS {ang} { \int_mod:nn { \dtexV{ang} - \l_dtex_turn_int + 1024 } {1024} } }
    \int_compare:nNnT { \dtexIn{right} } = {1}
      { \dtexS {ang} { \int_mod:nn { \dtexV{ang} + \l_dtex_turn_int } {1024} } }

    % refresh the view vectors: turning must be reflected before we move
    \dtexSetCamera

    \int_zero:N \l_dtex_dx_int \int_zero:N \l_dtex_dy_int
    \int_compare:nNnT { \dtexIn{forward} } = {1}
      {
        \int_add:Nn \l_dtex_dx_int { \l_dtex_dirx_int * \l_dtex_spd_int / 1024 }
        \int_add:Nn \l_dtex_dy_int { \l_dtex_diry_int * \l_dtex_spd_int / 1024 }
      }
    \int_compare:nNnT { \dtexIn{back} } = {1}
      {
        \int_sub:Nn \l_dtex_dx_int { \l_dtex_dirx_int * \l_dtex_spd_int / 1024 }
        \int_sub:Nn \l_dtex_dy_int { \l_dtex_diry_int * \l_dtex_spd_int / 1024 }
      }
    % strafing uses the camera's right vector, perpendicular to dir
    \int_compare:nNnT { \dtexIn{straferight} } = {1}
      {
        \int_sub:Nn \l_dtex_dx_int { \l_dtex_diry_int * \l_dtex_spd_int / 1024 }
        \int_add:Nn \l_dtex_dy_int { \l_dtex_dirx_int * \l_dtex_spd_int / 1024 }
      }
    \int_compare:nNnT { \dtexIn{strafeleft} } = {1}
      {
        \int_add:Nn \l_dtex_dx_int { \l_dtex_diry_int * \l_dtex_spd_int / 1024 }
        \int_sub:Nn \l_dtex_dy_int { \l_dtex_dirx_int * \l_dtex_spd_int / 1024 }
      }

    \bool_lazy_or:nnT
      { \int_compare_p:n { \l_dtex_dx_int != 0 } }
      { \int_compare_p:n { \l_dtex_dy_int != 0 } }
      {
        \dtexTryMove { \l_dtex_dx_int } { \l_dtex_dy_int }
        \dtexS {bob} { \int_mod:nn { \dtexV{bob} + 1 } {8} }
      }
  }

%% ------------------------------------------------------------------
%% Doors.  "use" acts on the cell one unit in front of the player.
%% ------------------------------------------------------------------
\cs_new_protected:Npn \dtexUseDoor
  {
    \int_compare:nNnT { \dtexIn{use} } = {1}
      {
        \int_set:Nn \l_dtex_lx_int { \dtexV{px} + \l_dtex_dirx_int }
        \int_set:Nn \l_dtex_ly_int { \dtexV{py} + \l_dtex_diry_int }
        \int_compare:nNnTF
          { \dtexCell { \dtexFloor{\l_dtex_lx_int} } { \dtexFloor{\l_dtex_ly_int} } } = {4}
          {
            \int_compare:nNnTF
              { \dtexDoorOpen { \dtexFloor{\l_dtex_lx_int} } { \dtexFloor{\l_dtex_ly_int} } } = {1}
              {
                \dtexSetDoor { \dtexFloor{\l_dtex_lx_int} } { \dtexFloor{\l_dtex_ly_int} } {0}
                \dtexPlaySound {DSDOROPN} {90} {0}
              }
              {
                \dtexSetDoor { \dtexFloor{\l_dtex_lx_int} } { \dtexFloor{\l_dtex_ly_int} } {1}
                \dtexPlaySound {DSDORCLS} {90} {0}
              }
          }
          { \dtexPlaySound {DSNOWAY} {70} {0} }
      }
  }

%% ------------------------------------------------------------------
%% Pickups.  Walk over a medkit or a clip to collect it.
%% ------------------------------------------------------------------
\cs_new_protected:Npn \dtexPickups
  {
    \int_step_inline:nnn {1} { \g_dtex_nent_int }
      {
        \bool_lazy_and:nnT
          { \int_compare_p:n { \dtexEV{##1}{type} > 1 } }
          { \int_compare_p:n { \dtexEV{##1}{st} != 5 } }
          {
            \int_set:Nn \l_dtex_relx_int { \dtexEV{##1}{x} - \dtexV{px} }
            \int_set:Nn \l_dtex_rely_int { \dtexEV{##1}{y} - \dtexV{py} }
            \dtexSqrtSet \l_dtex_ed_int
              { ( \l_dtex_relx_int * \l_dtex_relx_int
                  + \l_dtex_rely_int * \l_dtex_rely_int ) / 1024 }
            \int_compare:nNnT { \l_dtex_ed_int } < {512}
              {
                \int_compare:nNnTF { \dtexEV{##1}{type} } = {2}
                  {
                    \int_compare:nNnT { \dtexV{health} } < {100}
                      {
                        \dtexS {health} { \dtexMin { \dtexV{health} + 25 } {100} }
                        \dtexES {##1} {st} {5}
                        \dtexPlaySound {DSITEMUP} {100} {0}
                      }
                  }
                  {
                    \dtexS {ammo} { \dtexMin { \dtexV{ammo} + 10 } {200} }
                    \dtexES {##1} {st} {5}
                    \dtexPlaySound {DSITEMUP} {100} {0}
                  }
              }
          }
      }
  }

%% ------------------------------------------------------------------
%% Firing.  A hitscan pistol: the shot is a cylinder along the view axis,
%% so an actor is hit when it lies close to the aim line and in front of
%% the player, with nothing solid in between.  Nearest target wins.
%% ------------------------------------------------------------------
\cs_new_protected:Npn \dtexFire
  {
    \dtexS {fired} {0}
    \int_compare:nNnT { \dtexIn{fire} } = {1}
      {
        \int_compare:nNnTF { \dtexV{ammo} } > {0}
          {
            \dtexS {ammo} { \dtexV{ammo} - 1 }
            \dtexS {fired} {1}
            \dtexPlaySound {DSPISTOL} {100} {0}

            \int_set:Nn \l_dtex_best_int {0}
            \int_set:Nn \l_dtex_bestd_int {32000}
            \int_step_inline:nnn {1} { \g_dtex_nent_int }
              {
                \bool_lazy_and:nnT
                  { \int_compare_p:n { \dtexEV{##1}{type} = 1 } }
                  { \int_compare_p:n { \dtexEV{##1}{st} < 4 } }
                  {
                    \int_set:Nn \l_dtex_relx_int { \dtexEV{##1}{x} - \dtexV{px} }
                    \int_set:Nn \l_dtex_rely_int { \dtexEV{##1}{y} - \dtexV{py} }
                    % forward = distance along the aim line, lat = offset from it
                    \int_set:Nn \l_dtex_fwd_int
                      { ( \l_dtex_relx_int * \l_dtex_dirx_int
                          + \l_dtex_rely_int * \l_dtex_diry_int ) / 1024 }
                    \int_set:Nn \l_dtex_lat_int
                      { ( 0 - \l_dtex_relx_int * \l_dtex_diry_int
                          + \l_dtex_rely_int * \l_dtex_dirx_int ) / 1024 }
                    \bool_lazy_all:nT
                      {
                        { \int_compare_p:n { \l_dtex_fwd_int > 0 } }
                        { \int_compare_p:n { \l_dtex_fwd_int < \l_dtex_bestd_int } }
                        { \int_compare_p:n
                            { \dtexAbs{\l_dtex_lat_int} < \c_dtex_imp_radius_int } }
                      }
                      {
                        \dtexLOSSet \l_dtex_los_int { \dtexV{px} } { \dtexV{py} }
                                    { \dtexEV{##1}{x} } { \dtexEV{##1}{y} }
                        \int_compare:nNnT { \l_dtex_los_int } = {1}
                          {
                            \int_set:Nn \l_dtex_best_int {##1}
                            \int_set:Nn \l_dtex_bestd_int { \l_dtex_fwd_int }
                          }
                      }
                  }
              }
            \int_compare:nNnT { \l_dtex_best_int } > {0} { \dtexHurtActor }
          }
          { \dtexPlaySound {DSEMPTY} {70} {0} }
      }
  }

%% Apply pistol damage to the actor selected in \l_dtex_best_int.
\cs_new_protected:Npn \dtexHurtActor
  {
    \dtexRandSet \l_dtex_dmg_int {12}
    \int_add:Nn \l_dtex_dmg_int {10}
    \dtexES { \l_dtex_best_int } {hp}
      { \dtexEV{\l_dtex_best_int}{hp} - \l_dtex_dmg_int }
    \int_compare:nNnTF { \dtexEV{\l_dtex_best_int}{hp} } > {0}
      {
        \dtexES { \l_dtex_best_int } {st} {3}   % pain
        \dtexES { \l_dtex_best_int } {t}  {1}
        \dtexPlaySoundAt {DSPOPAIN}
          { \dtexEV{\l_dtex_best_int}{x} } { \dtexEV{\l_dtex_best_int}{y} } {95}
      }
      {
        \dtexES { \l_dtex_best_int } {st} {4}   % dying
        \dtexES { \l_dtex_best_int } {t}  {4}
        \dtexS {kills} { \dtexV{kills} + 1 }
        \dtexPlaySoundAt {DSPODTH1}
          { \dtexEV{\l_dtex_best_int}{x} } { \dtexEV{\l_dtex_best_int}{y} } {100}
      }
  }

%% ------------------------------------------------------------------
%% Actor behaviour.
%%   0 idle -> 1 chase -> 2 attack ; 3 pain ; 4 dying -> 5 gone
%% ------------------------------------------------------------------
\cs_new_protected:Npn \dtexActorThink #1
  {
    \int_set:Nn \l_dtex_relx_int { \dtexEV{#1}{x} - \dtexV{px} }
    \int_set:Nn \l_dtex_rely_int { \dtexEV{#1}{y} - \dtexV{py} }
    \dtexSqrtSet \l_dtex_ed_int
      { ( \l_dtex_relx_int * \l_dtex_relx_int
          + \l_dtex_rely_int * \l_dtex_rely_int ) / 1024 }

    \int_case:nnF { \dtexEV{#1}{st} }
      {
        {0}  % idle: wake when the player is close enough and visible
          {
            \int_compare:nNnT { \l_dtex_ed_int } < {9216}
              {
                \dtexLOSSet \l_dtex_los_int { \dtexV{px} } { \dtexV{py} }
                            { \dtexEV{#1}{x} } { \dtexEV{#1}{y} }
                \int_compare:nNnT { \l_dtex_los_int } = {1}
                  {
                    \dtexES {#1} {st} {1}
                    \dtexPlaySoundAt {DSBGSIT1} { \dtexEV{#1}{x} } { \dtexEV{#1}{y} } {90}
                  }
              }
          }
        {1}  % chase: close in, and swing when in range
          {
            \int_compare:nNnTF { \l_dtex_ed_int } < {1400}
              { \dtexES {#1} {st} {2} \dtexES {#1} {t} {2} }
              { \dtexActorStep {#1} }
          }
        {2}  % attack
          {
            \int_compare:nNnTF { \dtexEV{#1}{t} } > {0}
              { \dtexES {#1} {t} { \dtexEV{#1}{t} - 1 } }
              {
                \int_compare:nNnTF { \l_dtex_ed_int } < {1600}
                  {
                    \dtexRandSet \l_dtex_dmg_int {8}
                    \int_add:Nn \l_dtex_dmg_int {5}
                    \dtexS {health} { \dtexMax { \dtexV{health} - \l_dtex_dmg_int } {0} }
                    \dtexS {flash} {2}
                    \dtexPlaySoundAt {DSCLAW} { \dtexEV{#1}{x} } { \dtexEV{#1}{y} } {100}
                    \dtexPlaySound {DSPLPAIN} {100} {0}
                    \dtexES {#1} {t} {3}
                  }
                  { \dtexES {#1} {st} {1} }
              }
          }
        {3}  % pain: flinch for a frame, then resume the chase
          {
            \int_compare:nNnTF { \dtexEV{#1}{t} } > {0}
              { \dtexES {#1} {t} { \dtexEV{#1}{t} - 1 } }
              { \dtexES {#1} {st} {1} }
          }
        {4}  % dying: run out the death animation, then lie still
          {
            \int_compare:nNnTF { \dtexEV{#1}{t} } > {0}
              { \dtexES {#1} {t} { \dtexEV{#1}{t} - 1 } }
              { \dtexES {#1} {st} {5} }
          }
      }
      { }  % 5 = gone
  }

%% Step one actor toward the player, sliding on each axis independently.
\cs_new_protected:Npn \dtexActorStep #1
  {
    \int_compare:nNnT { \l_dtex_ed_int } > {0}
      {
        \int_set:Nn \l_dtex_dx_int
          { 0 - \l_dtex_relx_int * 110 / \dtexMax { \l_dtex_ed_int } {1} }
        \int_set:Nn \l_dtex_dy_int
          { 0 - \l_dtex_rely_int * 110 / \dtexMax { \l_dtex_ed_int } {1} }
        \int_set:Nn \l_dtex_lx_int { \dtexEV{#1}{x} + \l_dtex_dx_int }
        \int_compare:nNnF
          { \dtexSolid { \dtexFloor{\l_dtex_lx_int} } { \dtexFloor{\dtexEV{#1}{y}} } } > {0}
          { \dtexES {#1} {x} { \l_dtex_lx_int } }
        \int_set:Nn \l_dtex_ly_int { \dtexEV{#1}{y} + \l_dtex_dy_int }
        \int_compare:nNnF
          { \dtexSolid { \dtexFloor{\dtexEV{#1}{x}} } { \dtexFloor{\l_dtex_ly_int} } } > {0}
          { \dtexES {#1} {y} { \l_dtex_ly_int } }
      }
  }

%% ------------------------------------------------------------------
%% The tick.
%% ------------------------------------------------------------------
\cs_new_protected:Npn \dtexTick
  {
    \dtexAudioReset
    \dtexSetCamera
    \dtexS {frame} { \dtexV{frame} + 1 }
    \int_compare:nNnT { \dtexV{flash} } > {0} { \dtexS {flash} { \dtexV{flash} - 1 } }

    \int_compare:nNnTF { \dtexV{health} } > {0}
      {
        \dtexPlayerMove
        \dtexUseDoor
        \dtexPickups
        \dtexFire
        \int_step_inline:nnn {1} { \g_dtex_nent_int }
          {
            \int_compare:nNnT { \dtexEV{##1}{type} } = {1} { \dtexActorThink {##1} }
          }
      }
      {
        % dead: the only input that does anything is "use", to restart
        \dtexPlaySound {DSPLDETH} {100} {0}
        \int_compare:nNnT { \dtexIn{use} } = {1} { \dtexNewGame }
      }
    \dtexSetCamera
  }
