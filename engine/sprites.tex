%% DoomTex :: engine/sprites.tex
%% Actor and weapon artwork.
%%
%% Unlike the wall textures, these are drawn by hand: pattern arithmetic
%% will happily produce convincing brickwork but it will never produce a
%% recognisable imp.  Each sprite is 24x24, one character per pixel.
%%
%%   . transparent   k black     d dark red   m red      h orange
%%   w white         g mid grey  G lt grey    D dk grey  y brass
%%   b dk brown      t mid brown
\ExplSyntaxOn

\int_new:N \l_dtex_sp_u_int

%% 255 is the transparency sentinel; no palette entry uses it.
\cs_new:Npn \dtex_sp_pal:n #1
  {
    \str_case:nnF {#1}
      {
        {k}{0} {b}{1} {t}{2} {d}{4} {m}{5} {h}{6}
        {D}{7} {g}{8} {G}{9} {y}{14} {w}{15}
      } {255}
  }

%% \dtexSpriteRow {sprite id} {row} {24 characters}
\cs_new_protected:Npn \dtexSpriteRow #1#2#3
  {
    \int_zero:N \l_dtex_sp_u_int
    \tl_map_inline:nn {#3}
      {
        \cs_gset:cpx
          { dtex_sp_ \int_eval:n {#1} _ \int_use:N \l_dtex_sp_u_int _ \int_eval:n {#2} : }
          { \dtex_sp_pal:n {##1} }
        \int_incr:N \l_dtex_sp_u_int
      }
  }

%% Expandable texel fetch; off-sprite reads as transparent.
\cs_new:Npn \dtexSpTexel #1#2#3
  { \cs_if_exist_use:cF
      { dtex_sp_ \int_eval:n {#1} _ \int_eval:n {#2} _ \int_eval:n {#3} : } {255} }

\int_const:Nn \c_dtex_sprite_dim_int {24}
